```python
import os
import re

import sqlalchemy as sa

TEST_MARIADB_URL = os.environ.get("TEST_MARIADB_URL")
# GitHub Actions ustawia CI=true i uruchamia MariaDB (ci.yml): tam brak TEST_MARIADB_URL to błąd konfiguracji,
# a nie powód do pominięcia — test ma się wywrócić w fixture mariadb_url, nie zniknąć jako „skipped”.
needs_mariadb = pytest.mark.skipif(
    not TEST_MARIADB_URL and not os.environ.get("CI"),
    reason="test na prawdziwej MariaDB — ustaw TEST_MARIADB_URL (pusta baza testowa)")


def require_test_database(url: str) -> None:
    """Testy czyszczą całą bazę — odmowa, gdy nazwa bazy w URL nie zawiera słowa „test” jako osobnego członu
    (xp_test, test-db, test — tak; latest, contest, baza demo — nie)."""
    name = (sa.make_url(url).database or "").lower()
    if "test" not in re.split(r"[^a-z0-9]+", name):
        pytest.fail("TEST_MARIADB_URL musi wskazywać bazę testową (nazwa bazy zawiera człon „test”) — testy MariaDB "
                    "usuwają wszystkie tabele.", pytrace=False)


def wipe_mariadb(url: str) -> None:
    """Czyści bazę testową: tabele aplikacji (jak `db.drop_all()`) oraz `alembic_version`."""
    engine = sa.create_engine(url)
    try:
        with engine.begin() as conn:
            conn.execute(sa.text("SET FOREIGN_KEY_CHECKS = 0"))  # kolejność kasowania bez znaczenia, także po awarii
            db.metadata.drop_all(conn)
            conn.execute(sa.text("DROP TABLE IF EXISTS alembic_version"))
    finally:
        engine.dispose()


@pytest.fixture
def mariadb_url():
    url = os.environ.get("TEST_MARIADB_URL")
    if not url:
        pytest.fail("Brak TEST_MARIADB_URL, a CI=true — w CI testy MariaDB są obowiązkowe (patrz .github/workflows/ci.yml).",
                    pytrace=False)
    require_test_database(url)
    wipe_mariadb(url)
    yield url
    wipe_mariadb(url)  # także po nieudanym teście
```

```python
import functools
import struct
from pathlib import Path

SCREENSHOTS_DIR = Path(__file__).resolve().parent.parent / "static" / "screenshots"
PNG_SIGNATURE = b"\x89PNG\r\n\x1a\n"


@functools.lru_cache(maxsize=64)
def screenshot_size(file: str) -> tuple[int, int] | None:
    """(szerokość, wysokość) pliku PNG z app/static/screenshots/ odczytane z nagłówka IHDR — atrybuty width/height
    obrazka rezerwują miejsce przed doczytaniem (loading="lazy") i układ nie skacze. None, gdy pliku nie ma
    albo nie jest PNG (strona ma się wyrenderować bez wymiarów, nie wywrócić)."""
    try:
        header = (SCREENSHOTS_DIR / file).read_bytes()[:24]
    except OSError:
        return None
    if len(header) < 24 or not header.startswith(PNG_SIGNATURE) or header[12:16] != b"IHDR":
        return None
    width, height = struct.unpack(">II", header[16:24])
    return width, height
```

```python
    @app.cli.command("set-admin-password")
    @click.option("--email", default=None, help="E-mail konta (domyślnie ADMIN_EMAIL z konfiguracji).")
    @click.password_option("--password", prompt="Nowe hasło", confirmation_prompt=True, hide_input=True,
                           help="Nowe hasło; pominięte = pytanie interaktywne (hasła nie widać w liście procesów).")
    def set_admin_password_cmd(email, password):
        """Ustawia hasło administratora (tworzy konto, gdy go nie ma) i wylogowuje jego sesje."""
        email = (email or app.config.get("ADMIN_EMAIL") or "").strip().lower()
        if not email:
            click.echo("Podaj --email albo ustaw ADMIN_EMAIL.", err=True)
            raise SystemExit(1)
        if password_is_weak(password):
            click.echo("Hasło musi mieć co najmniej 10 znaków i nie może być hasłem przykładowym.", err=True)
            raise SystemExit(1)
        user = models.User.query.filter_by(email=email).first()
        if user is None:
            user = models.User(email=email, role="admin")
            db.session.add(user)
        user.set_password(password)
        user.rotate_session_token()
        db.session.commit()
        click.echo(f"Hasło ustawione dla {email}; zalogowane sesje tego konta zostały wylogowane.")
```
